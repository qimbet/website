\newcommand{\lang}[2]{%
	\noindent
	\makebox[0pt][l]{#1}% Print language name without taking horizontal space
	\hspace{75mm}% Shift the line right
	\begin{tikzpicture}[baseline={(0,-0.5ex)}]
		% Scale input #2 (range 1-5) to match 7-unit line
		\pgfmathsetmacro{\scaledlevel}{(#2 - 1) * (7/4)} 
		
		\filldraw[black] (\scaledlevel,0) circle (2.2pt);  % Scaled fluency level
		\draw[thick] (0,0) -- (7,0);                       % Longer base line
		\draw[thick] (0,0.1) -- (0,-0.1);                  % Tick at Basic
		\draw[thick] (7,0.1) -- (7,-0.1);                  % Tick at Fluent
		\node[above] at (0,0.08) {\scriptsize Basic};
		\node[above] at (7,0.08) {\scriptsize Fluent};
	\end{tikzpicture}
}
