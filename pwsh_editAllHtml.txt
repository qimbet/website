$folder = Get-Location

# Code to insert at the top (right after <body>)
$codeAtBodyStart = @"
 <div id="sidebar-tab">⟩</div>
<div id='sidebar-overlay'></div>
"@



# Code to insert at the bottom (right before </body>)
$codeAtBodyEnd = @"
<script>
const sidebar = document.getElementById('sidebar');
const overlay = document.getElementById('sidebar-overlay');
toggleBtn.addEventListener('click', () => {
    sidebar.classList.toggle('open');
    overlay.classList.toggle('active');
});
overlay.addEventListener('click', () => {
    sidebar.classList.remove('open');
    overlay.classList.remove('active');
});
</script>
"@



Get-ChildItem $folder -Filter *.html -Recurse | ForEach-Object {
    $content = Get-Content $_.FullName -Raw

    # Insert after opening <body> tag
    $content = $content -replace '(?i)(<body[^>]*>)', "`$1`r`n$codeAtBodyStart"

    # Insert before closing </body> tag
    $content = $content -replace '(?i)</body>', "$codeAtBodyEnd`r`n</body>"

    Set-Content $_.FullName -Value $content
}
